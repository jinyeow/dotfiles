# This file will outline how I install UEFI Arch Linux
# 17 July 2015
# These are taken from the following:
# 1. Arch Linux Wiki Beginner's Guide
# ---> https://wiki.archlinux.org/index.php/Beginners%27_guide
# 2. Installing Arch on a UEFI GPT system | Jorisvandijk.com
# ---> http://jorisvandijk.com/2014/installing-arch-on-a-uefi-gpt-system/
# 3. Install Arch Linux on virtualbox - the nuts and bolts
# ---> http://dwheelerau.com/2014/07/25/install-arch-linux-on-virtualbox-the-nuts-and-bolts-pt1/
# ---> http://dwheelerau.com/2014/07/25/install-arch-linux-on-virtualbox-the-nuts-and-bolts-pt2/

# Check for UEFI variables:
efivar -l

# Check for Secure Boot:
od -An -t u1 /sys/firmware/efi/efivars/SecureBoot-XXXXXXXX-XXXX-XXXX-XXXX-XXXXXXXXXXXX

# Setup wifi connection
# Find/enter SSID and enter Password/Key
wifi-menu

# Check connection
ping -c 3 www.google.com

# Partition /dev/sda. use gdisk for UEFI boot
gdisk /dev/sda

# create new GUID partition table and destroy everything on disk
(gdsik) o

# create new partition
(gdisk) n

# create the following partitions:
sda1
1024M (maybe 512M will do?)
GUID=EF00
/boot

sda2
20-25G
GUID=8300
/

sda3
~G (remaining disk space)
GUID=8300
/home

# show partition table as it is now
(gdisk) p

# write changes and quit
(gdisk) w

# create filesystems
mkfs.ext4 /dev/sda2
mkfs.ext4 /dev/sda3
mkfs.fat -F32 /dev/sda1

# mount filesystems
mount /dev/sda2 /mnt
mkdir /mnt/boot
mkdir /mnt/home
mount /dev/sda1 /mnt/boot
mount /dev/sda3 /mnt/home

# install base packages/system
pacstrap /mnt base base-devel
*NB: if the '-i' flag is used pacstrap will ask for confirmations

# set up fstab
genfstab -U -p /mnt >> /mnt/etc/fstab

# check the fstab table
vi /mnt/etc/fstab

# configure the system
arch-chroot /mnt [/bin/bash]

# create locale file
# remove the "#" from in front of the locale(s) you need,
# e.g. en_AU.UTF-8 for me.
vi /etc/locale.gen

# save the file and generate the locales
locale-gen

# locale.conf
echo LANG=en_AU.UTF-8 > /etc/locale.conf
export LANG=en_AU.UTF-8

# set up local time
ln -s /usr/share/zoneinfo/Australia/Sydney /etc/localtime

# set the clock to UTC
hwclock --systohc --utc

# set up hostname
echo [hostname] > /etc/hostname
vi /etc/hosts (and add the [hostname] to the end of each line of the table)

# set up wireless/network
pacman -S iw wpa_supplicant
pacman -S dialog
pacman -S wpa_actiond
systemctl enable netctl-auto@[interface_name].service
*NB: use wifi-menu after finishing installation and rebooting

# set root password
passwd

# install bootloader
mount -t efivarfs efivarfs /sys/firmware/efi/efivars
pacman -S grub

# check if you have GPT and an ESP
parted /dev/sda print
*NB: this step is optional. do as root

# bootloader "systemd-boot" (for now I prefer grub)
bootctl --path=/boot install
# setup grub bootloader
arch-chroot /mnt /bin/bash
*NB: if not in arch-chroot already
grub-install --target=x86_64-efi --efi-directory=$esp --bootloader-id=grub --recheck
*NB: where $esp is the efi partition. usually (and for my purposes) /boot.

# make grub config
grub-mkconfig -o $esp/grub/grub.cfg
*NB: again where $esp is the efi partition (/boot)

# setup Secure Boot for Arch -- on second thoughts this is more trouble than it's worth
# just disable Secure Boot
# https://www.reddit.com/r/archlinux/comments/2x10ip/arch_linux_and_uefi/cowdcdk
# DO THIS BEFORE EXITING CHROOT AND INSTALLATION ENVIRONMENT
pacman -S prebootloader
pacman -S efibootmgr
# copy files from /usr/lib/prebootloader to /boot/EFI/gummiboot
# rename gummiboot/grub file to loader.efi
# use efibootmgr to add UEFI boot entry pointing to PreLoader.efi
# steps to do so outlined here: https://wiki.archlinux.org/index.php/Unified_Extensible_Firmware_Interface#efibootmgr
# replace 'refind_x64.efi' with 'PreLoader.efi' and make sure path is correct
efibootmgr -c -d /dev/sda -p 1 -l /EFI/refind/refind_x64.efi -L "Arch Linux Secure Boot"
# verify the efibootmgr command worked by running
efibootmgr -v

# unmount partitions and reboot
exit
umount -R /mnt
reboot (remove installation media so device loads os from memory and not from usb)

# log in as root with root password
# create a user
useradd -m -g users -G wheel -s [/bin/bash or /bin/zsh] [username]

# create a password for user
passwd [username]

# add user to the sudoers group
echo '[username] ALL=(ALL) NOPASSWD:ALL' >> /etc/sudoers
*NB: "NOPASSWD:ALL" lets you do sudo commands without having to enter a password.
This is potentially dangerous!!

# sync pacman database
pacman -Syy

# add aur repo to grab yaourt
vi /etc/pacman.conf
*NB: add the following to the end of pacman.conf
[archlinuxfr]
SigLevel = Never
Server = http://repo.archlinux.fr/$arch

# install xorg
pacman -S xorg-server xorg-server-utils xorg-xinit mesa

# video card drivers
# intel
pacman -S xf86-video-intel

# input drivers
pacman -S xf86-input-evdev

# trackpad
pacman -S xf86-input-synaptics

# install window manager
pacman -S bspwm or i3wm

# exit root
exit

# start x server
startx
